## Usage

```tsx
import {
  Checkbox,
  Field,
  FieldHelp,
  Input,
  Label,
  Select,
  Switch,
  Textarea,
} from "@/components/ui/form-fields"
```

```tsx
<Field>
  <Label htmlFor="email">Email</Label>
  <Input id="email" type="email" placeholder="you@example.com" />
  <FieldHelp>We will never share your email.</FieldHelp>
</Field>
```

## Examples

### Textarea

```tsx
<Field>
  <Label htmlFor="bio">About me</Label>
  <Textarea id="bio" placeholder="Tell us about yourself" />
</Field>
```

### Switch

The switch is uncontrolled by default. Pass `checked` and `onCheckedChange` to control it.

```tsx
const [enabled, setEnabled] = React.useState(false)

<Switch checked={enabled} onCheckedChange={setEnabled} aria-label="Enable" />
```

### Checkbox

```tsx
<label className="flex items-center gap-2">
  <Checkbox defaultChecked />
  <span>Remember me</span>
</label>
```
