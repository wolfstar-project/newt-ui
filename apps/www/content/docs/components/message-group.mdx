## Usage

```tsx
import {
  MessageGroup,
  MessageGroupBody,
  MessageGroupHeader,
  MessageGroupAuthor,
  MessageGroupBotTag,
  MessageGroupTime,
  MessageGroupContent,
  MessageGroupReactions,
} from "@/components/ui/message-group"
```

```tsx
<MessageGroup>
  <Avatar>U</Avatar>
  <MessageGroupBody>
    <MessageGroupHeader>
      <MessageGroupAuthor>username</MessageGroupAuthor>
      <MessageGroupBotTag />
      <MessageGroupTime>Today at 14:23</MessageGroupTime>
    </MessageGroupHeader>
    <MessageGroupContent>Message content goes here.</MessageGroupContent>
    <MessageGroupReactions>
      <Reaction>
        <span aria-hidden="true">👍</span>
        <ReactionCount>1</ReactionCount>
      </Reaction>
    </MessageGroupReactions>
  </MessageGroupBody>
</MessageGroup>
```
