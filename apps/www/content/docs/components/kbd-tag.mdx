## Usage

```tsx
import { Kbd } from "@/components/ui/kbd-tag"
```

```tsx
<Kbd>Ctrl</Kbd>
<Kbd>K</Kbd>
```

## Examples

### Inline in text

```tsx
<p className="text-sm text-newt-text-muted">
```
