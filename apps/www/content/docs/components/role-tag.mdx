## Usage

```tsx
import { RoleTag, RoleTagDot } from "@/components/ui/role-tag"
```

```tsx
<RoleTag>
  <RoleTagDot />
  Role name
</RoleTag>
```

## Examples

### Role colour

Pass `color` to tint the tag text; the dot follows via `currentColor`.

```tsx
<RoleTag color="#5865f2">
  <RoleTagDot />
  Admin
</RoleTag>
```
