## Usage

```tsx
import {
  StageBanner,
  StageBannerDot,
  StageBannerAction,
  StageBannerClose,
} from "@/components/ui/stage-banner"
```

```tsx
<StageBanner>
  <StageBannerDot />
  Announcement text goes here.
  <StageBannerAction>Action</StageBannerAction>
</StageBanner>
```

## Examples

### Dismissible (neutral)

```tsx
<StageBanner variant="neutral" onOpenChange={(open) => console.log(open)}>
  <StageBannerDot status="online" />
  Status message goes here.
  <StageBannerClose className="text-newt-text-muted hover:text-newt-text-primary" />
</StageBanner>
```

`StageBannerClose` hides the banner when clicked. Pass `open` / `onOpenChange` to control visibility yourself.
