## Usage

```tsx
import { ScrollArea, scrollbarClassName } from "@/components/ui/scrollbar"
```

```tsx
<ScrollArea className="max-h-[240px] overflow-y-auto">
  {/* scrollable content */}
</ScrollArea>
```

## Examples

### Apply to an existing element

Spread `scrollbarClassName` onto any element you already control.

```tsx
<div className={cn("max-h-60 overflow-y-auto", scrollbarClassName)}>...</div>
```
