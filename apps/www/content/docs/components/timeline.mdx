## Usage

```tsx
import {
  Timeline,
  TimelineItem,
  TimelineDot,
  TimelineContent,
  TimelineTitle,
  TimelineMeta,
} from "@/components/ui/timeline"
```

```tsx
<Timeline>
  <TimelineItem>
    <TimelineDot variant="brand">↻</TimelineDot>
    <TimelineContent>
      <TimelineTitle>Event title</TimelineTitle>
      <TimelineMeta>Detail · timestamp</TimelineMeta>
    </TimelineContent>
  </TimelineItem>
  <TimelineItem>
    <TimelineDot variant="success">✓</TimelineDot>
    <TimelineContent>
      <TimelineTitle>Event title</TimelineTitle>
      <TimelineMeta>Detail · timestamp</TimelineMeta>
    </TimelineContent>
  </TimelineItem>
</Timeline>
```
