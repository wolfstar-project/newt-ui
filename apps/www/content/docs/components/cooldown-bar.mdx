## Usage

```tsx
import {
  CooldownBar,
  CooldownBarLabel,
  CooldownBarTrack,
  CooldownBarValue,
} from "@/components/ui/cooldown-bar"
```

```tsx
<CooldownBar>
  <CooldownBarLabel>
    <span>Command cooldown</span>
    <CooldownBarValue>3.2s</CooldownBarValue>
  </CooldownBarLabel>
  <CooldownBarTrack value={40} aria-label="Command cooldown" />
</CooldownBar>
```

## Examples

### Ready state

Pass `ready` to switch the fill colour from idle yellow to online green once the cooldown has elapsed.

```tsx
<CooldownBarTrack value={100} ready aria-label="Command cooldown" />
```
