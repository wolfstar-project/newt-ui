## Usage

```tsx
import {
  SelectMenu,
  SelectMenuDivider,
  SelectMenuLabel,
  SelectMenuOption,
  SelectMenuSearch,
} from "@/components/ui/select-menu"
```

```tsx
<SelectMenu aria-label="Select an option">
  <SelectMenuSearch placeholder="Search…" aria-label="Search options" />
  <SelectMenuLabel>Options</SelectMenuLabel>
  <SelectMenuOption selected>Option one</SelectMenuOption>
  <SelectMenuOption>Option two</SelectMenuOption>
  <SelectMenuDivider />
  <SelectMenuOption>Option three</SelectMenuOption>
</SelectMenu>
```

## Examples

### Selection state

`SelectMenuOption` is presentational: pass `selected` to show the check mark and handle `onClick` to update your own state. The menu itself is a `role="listbox"`; options set `aria-selected` automatically.
