## Usage

```tsx
import { Reaction } from "@/components/ui/reaction"
```

```tsx
<Reaction emoji="👍" count={12} defaultActive />
<Reaction emoji="🚀" count={4} />
```

## Examples

### Controlled

```tsx
const [active, setActive] = React.useState(false)

<Reaction
  emoji="🔥"
  count={active ? 8 : 7}
  active={active}
  onActiveChange={setActive}
/>
```
