## Usage

```tsx
import {
  Attachment,
  AttachmentIcon,
  AttachmentMeta,
  AttachmentName,
  AttachmentSize,
} from "@/components/ui/attachment"
import { Button } from "@/components/ui/button"
```

```tsx
<Attachment>
  <AttachmentIcon>PDF</AttachmentIcon>
  <AttachmentMeta>
    <AttachmentName>report.pdf</AttachmentName>
    <AttachmentSize>1.2 MB</AttachmentSize>
  </AttachmentMeta>
  <Button variant="icon" aria-label="Download file">
    ⬇
  </Button>
</Attachment>
```
