## Usage

```tsx
import { Mention } from "@/components/ui/mention"
```

```tsx
<Mention variant="user">@username</Mention>
<Mention variant="channel">#channel</Mention>
<Mention variant="role">@RoleName</Mention>
```

## Examples

### Variants

`variant` accepts `"user"` (default), `"channel"` or `"role"`.
