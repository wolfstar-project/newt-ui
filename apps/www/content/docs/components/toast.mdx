## Usage

```tsx
import {
  Toast,
  ToastIcon,
  ToastContent,
  ToastTitle,
  ToastDescription,
  ToastClose,
} from "@/components/ui/toast"
```

```tsx
<Toast
  variant="success"
  duration={5000}
  onOpenChange={(open) => !open && remove()}
>
  <ToastIcon />
  <ToastContent>
    <ToastTitle>Success title</ToastTitle>
    <ToastDescription>Description text.</ToastDescription>
  </ToastContent>
  <ToastClose />
</Toast>
```

## Examples

### Error

```tsx
<Toast variant="error">
  <ToastIcon />
  <ToastContent>
    <ToastTitle>Something went wrong</ToastTitle>
    <ToastDescription>Please try again.</ToastDescription>
  </ToastContent>
</Toast>
```

### Persistent

Pass `duration={0}` to disable auto-dismiss; the toast then only closes via `ToastClose` or the controlled `open` prop.
