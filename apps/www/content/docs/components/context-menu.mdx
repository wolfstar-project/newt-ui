## Usage

```tsx
import {
  ContextMenu,
  ContextMenuDivider,
  ContextMenuItem,
  ContextMenuShortcut,
} from "@/components/ui/context-menu"
```

```tsx
<ContextMenu>
  <ContextMenuItem>Action one</ContextMenuItem>
  <ContextMenuItem>
    Action two
    <ContextMenuShortcut>Ctrl+K</ContextMenuShortcut>
  </ContextMenuItem>
  <ContextMenuDivider />
  <ContextMenuItem variant="danger">Delete</ContextMenuItem>
</ContextMenu>
```

## Examples

### Danger item
