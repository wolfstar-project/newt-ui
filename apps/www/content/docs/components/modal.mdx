## Usage

```tsx
import {
  Modal,
  ModalBody,
  ModalClose,
  ModalFooter,
  ModalHeader,
} from "@/components/ui/modal"
```

```tsx
const [open, setOpen] = React.useState(false)

<Modal open={open} onOpenChange={setOpen}>
  <ModalHeader>Modal title</ModalHeader>
  <ModalBody>Modal body content goes here.</ModalBody>
  <ModalFooter>
    <ModalClose asChild>
      <Button variant="secondary">Cancel</Button>
    </ModalClose>
    <Button variant="danger">Confirm</Button>
  </ModalFooter>
</Modal>
```

## Behaviour

- `open` controls visibility; `onOpenChange(false)` is called when the user presses Escape, clicks the overlay, or activates a `ModalClose`.
- `ModalHeader` is wired to the dialog via `aria-labelledby` automatically.
- `ModalClose` renders a plain `<button>` by default; pass `asChild` to attach the close behaviour to your own element.
