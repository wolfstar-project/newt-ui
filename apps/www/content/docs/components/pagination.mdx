## Usage

```tsx
import {
  LoadMore,
  Pagination,
  PaginationButton,
  PaginationEllipsis,
} from "@/components/ui/pagination"
```

```tsx
<Pagination>
  <PaginationButton aria-label="Previous page">&lsaquo;</PaginationButton>
  <PaginationButton active>1</PaginationButton>
  <PaginationButton>2</PaginationButton>
  <PaginationEllipsis />
  <PaginationButton>10</PaginationButton>
  <PaginationButton aria-label="Next page">&rsaquo;</PaginationButton>
</Pagination>
```

## Examples

### Load more

`LoadMore` centres its content between two hairline rules, for infinite-scroll style lists.

```tsx
<LoadMore>
  <Button variant="secondary" size="sm">
    Load more
  </Button>
</LoadMore>
```
